# LuckPerms groups for Two Lives, as console commands (no leading slash). Source of truth for the group layout;
# apply on the server with the loop in docs/server.md "Permissions (LuckPerms)" (one RCON call per line; safe to run again).
# Order: default < linked < modder < staff, each inheriting the one before. The Discord roles of the same name are
# only labels: sdlink does not set LuckPerms groups, discord/roles.js or an op runs `lp user <name> parent add <group>`.
#
# What these nodes reach: `minecraft.command.<name>` (vanilla commands, via LuckPerms' command hook), `worldedit.*`,
# FTB XMod Compat permission checks and `luckperms.*`. The pack's own /ttl, /ttlbot and /tensura edit use
# `hasPermission(2)` (op level 2) and are NOT gated by these groups: keep builders/staff opped at level 2 or move
# those commands to a permission node first (see docs/server.md).
# Verify in game before relying on a node: `lp user <name> permission check <node>` and try the command as that user.

# default: every player (LuckPerms creates it). No extra nodes: vanilla level-0 commands already work.
lp group default setweight 0
lp group default setdisplayname Player

# linked: Minecraft account linked to Discord (sdlink /verify). Same rights as default; the group exists so
# quests, claims or perks can key on it (e.g. `ftbchunks.max_claimed` meta) without a code change.
lp creategroup linked
lp group linked setweight 10
lp group linked setdisplayname Linked
lp group linked parent add default

# modder: pack contributors and builders who test in game (also the trusted testers).
lp creategroup modder
lp group modder setweight 50
lp group modder setdisplayname Modder
lp group modder parent add linked
lp group modder permission set worldedit.* true
lp group modder permission set minecraft.command.gamemode true
lp group modder permission set minecraft.command.tp true
lp group modder permission set minecraft.command.teleport true
lp group modder permission set minecraft.command.time true
lp group modder permission set minecraft.command.weather true
lp group modder permission set minecraft.command.give true
lp group modder permission set minecraft.command.effect true
lp group modder permission set minecraft.command.summon true
lp group modder permission set minecraft.command.setblock true
lp group modder permission set minecraft.command.fill true
lp group modder permission set minecraft.command.clone true
lp group modder permission set minecraft.command.kill true

# staff: owner and moderators. Moderation, whitelist, FTB admin and LuckPerms itself.
lp creategroup staff
lp group staff setweight 100
lp group staff setdisplayname Staff
lp group staff parent add modder
lp group staff permission set minecraft.command.kick true
lp group staff permission set minecraft.command.ban true
lp group staff permission set minecraft.command.ban-ip true
lp group staff permission set minecraft.command.pardon true
lp group staff permission set minecraft.command.pardon-ip true
lp group staff permission set minecraft.command.whitelist true
lp group staff permission set minecraft.command.gamerule true
lp group staff permission set minecraft.command.reload true
lp group staff permission set minecraft.command.save-all true
lp group staff permission set minecraft.command.difficulty true
lp group staff permission set luckperms.* true
